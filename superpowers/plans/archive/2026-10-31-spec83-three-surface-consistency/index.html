def _meta_regions(chapter_text: str) -> list[tuple[str, int, int]]:
    """(region_text, start, end) —— 位置法排尾注需要 match 绝对位置。"""
    regions: list[tuple[str, int, int]] = []

        regions.append((m.group(0), m.start(), m.end()))
    return regions
