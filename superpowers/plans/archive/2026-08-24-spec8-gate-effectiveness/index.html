def _entries(data: dict) -> list[dict]:
    # gate_manifest.record_gate_result 的存储形状以实际实现为准；若为嵌套
    # {phase: {chapter: [...]}} 则递归展平
    out: list[dict] = []
    def walk(v: object) -> None:
        if isinstance(v, dict) and "gate" in v:
            out.append(v)
        elif isinstance(v, dict):
            for x in v.values():
                walk(x)
        elif isinstance(v, list):
            for x in v:
                walk(x)
    walk(data)
    return out
